Complete Bookmarklet Source Code

Self-contained, zero-dependency JavaScript snippet that powers Clauree.

javascript:(function(){
    if(window.__claureeLoaded) { alert('Clauree is already active on this page.'); return; }
    window.__claureeLoaded = true;

    // Core Clauree State
    const state = {
        msgCount: 3,
        maxMsgs: 10,
        resetTime: Date.now() + (3*3600 + 24*60)*1000,
        queuedPrompt: null
    };

    // Inject Widget Overlay into DOM
    const container = document.createElement('div');
    container.id = 'clauree-injected-root';
    container.innerHTML = `
        <div style="position:fixed; bottom:80px; right:20px; z-index:999999; width:320px; background:#282524; color:#EAE6DF; border:1px solid #3D3937; border-radius:12px; font-family:sans-serif; box-shadow:0 10px 25px rgba(0,0,0,0.5); font-size:12px; overflow:hidden;">
            <div style="padding:10px 14px; background:#1C1A19; display:flex; justify-between; align-items:center; border-bottom:1px solid #3D3937;">
                <span style="font-weight:bold; color:#D95D39;">⚡ Clauree Helper</span>
                <span id="clauree-timer" style="font-family:monospace; color:#fbbf24;">03h 24m</span>
            </div>
            <div style="padding:12px;">
                <div style="display:flex; justify-content:space-between; margin-bottom:4px; font-size:11px;">
                    <span>Quota Usage</span>
                    <span id="clauree-usage-text">3 / 10 Msgs</span>
                </div>
                <div style="width:100%; height:6px; background:#1C1A19; border-radius:3px; overflow:hidden; margin-bottom:10px;">
                    <div id="clauree-bar" style="width:30%; height:100%; background:#D95D39; transition:width 0.3s;"></div>
                </div>
                <div style="display:flex; gap:6px;">
                    <button id="clauree-btn-export" style="flex:1; padding:6px; background:#3D3937; color:#fff; border:none; border-radius:6px; cursor:pointer;">Export Context</button>
                    <button id="clauree-btn-close" style="padding:6px 10px; background:transparent; color:#8C857B; border:none; cursor:pointer;">✕</button>
                </div>
            </div>
        </div>
    `;
    document.body.appendChild(container);

    // Passive DOM Monitor for Rate Limits
    const observer = new MutationObserver(() => {
        if(document.body.innerText.includes('Message limit reached') || document.body.innerText.includes('429')) {
            alert('Clauree Emergency Fallback: Quota Block Detected! Open Clauree Menu to export context.');
        }
    });
    observer.observe(document.body, { childList: true, subtree: true });
})();
